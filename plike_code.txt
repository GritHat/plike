function bubble_sort(in x: integer, in n: integer, in arr[n]: array of integer): integer

    var ordered : logical
    j := n-1
    while j > 0 and not ordered do
        ordered := true
        i := 0
        while i < j do
            if arr[i] > arr[i+1] then
                arr[i] = arr[i] ^ arr[i+1]
                arr[i+1] = arr[i+1] ^ arr[i]
                arr[i] = arr[i] ^ arr[i+1]
